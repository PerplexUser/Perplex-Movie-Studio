cmake_minimum_required(VERSION 3.24)
project(PerplexMovieStudio VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(PMS_BUILD_GUI "Build the desktop application and media services" ON)
set(PMS_QT_VERSION "6.8.3" CACHE STRING "Reviewed Qt SDK version")
add_library(pms_domain src/domain/Project.cpp src/commands/Editor.cpp)
target_include_directories(pms_domain PUBLIC src)
if(MSVC)
  target_compile_options(pms_domain PUBLIC /W4 /permissive- /utf-8)
else()
  target_compile_options(pms_domain PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(domain_tests tests/unit/domain_tests.cpp)
  target_link_libraries(domain_tests PRIVATE pms_domain)
  add_test(NAME domain COMMAND domain_tests)
endif()
if(PMS_BUILD_GUI)
  find_package(Qt6 ${PMS_QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Concurrent Widgets Multimedia MultimediaWidgets Test LinguistTools)
  qt_standard_project_setup()
  add_library(pms_services
    src/persistence/ProjectStore.cpp
    src/media/MediaTools.cpp src/media/ProbeJob.cpp
    src/render/Composition.cpp src/render/VisualAssets.cpp src/export/RenderJob.cpp)
  target_include_directories(pms_services PUBLIC src)
  target_link_libraries(pms_services PUBLIC pms_domain Qt6::Core Qt6::Gui Qt6::Concurrent)
  add_library(pms_ui src/app/Session.cpp src/ui/MainWindow.cpp src/ui/MainWindowFeatures.cpp src/ui/PreviewWidget.cpp src/ui/Timeline.cpp src/ui/ExportDialog.cpp src/ui/MediaBin.cpp)
  target_link_libraries(pms_ui PUBLIC pms_services Qt6::Widgets Qt6::Multimedia)
  target_compile_definitions(pms_ui PUBLIC PMS_VERSION="${PROJECT_VERSION}-dev")
  qt_add_executable(PerplexMovieStudio src/app/main.cpp)
  target_link_libraries(PerplexMovieStudio PRIVATE pms_ui)
  qt_add_translations(PerplexMovieStudio TS_FILES translations/perplex_de_DE.ts
    RESOURCE_PREFIX /i18n LUPDATE_OPTIONS -no-obsolete)
  if(WIN32)
    target_sources(PerplexMovieStudio PRIVATE src/app/windows.rc)
    target_include_directories(PerplexMovieStudio PRIVATE assets)
    set_target_properties(PerplexMovieStudio PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  install(TARGETS PerplexMovieStudio RUNTIME DESTINATION .)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md DESTINATION .)
  install(DIRECTORY docs/ DESTINATION docs PATTERN specification EXCLUDE)
  if(BUILD_TESTING)
    add_executable(service_tests tests/integration/service_tests.cpp)
    target_link_libraries(service_tests PRIVATE pms_services Qt6::Test)
    add_custom_command(TARGET service_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures $<TARGET_FILE_DIR:service_tests>/fixtures)
    add_test(NAME services COMMAND service_tests)
    set_tests_properties(services PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 300)
    add_test(NAME ui_smoke COMMAND PerplexMovieStudio --self-test-ui)
    set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 25)
    add_executable(preview_tests tests/integration/preview_tests.cpp)
    target_link_libraries(preview_tests PRIVATE pms_ui Qt6::Test)
    add_custom_command(TARGET preview_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures $<TARGET_FILE_DIR:preview_tests>/fixtures)
    add_test(NAME preview_playback COMMAND preview_tests)
    set_tests_properties(preview_playback PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 25)
    add_executable(feature_tests tests/integration/feature_tests.cpp)
    target_link_libraries(feature_tests PRIVATE pms_services Qt6::Test)
    add_test(NAME visual_features COMMAND feature_tests)
    set_tests_properties(visual_features PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 300)
    add_executable(window_tests tests/integration/window_tests.cpp)
    target_link_libraries(window_tests PRIVATE pms_ui Qt6::Test)
    add_dependencies(window_tests PerplexMovieStudio)
    add_test(NAME editor_workflow COMMAND window_tests)
    set_tests_properties(editor_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
  endif()
endif()
